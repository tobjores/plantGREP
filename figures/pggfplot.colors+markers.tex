%%% define extra markers
% filled circle without border
\pgfdeclareplotmark{solido}{%
	\pgfpathcircle{\pgfpointorigin}{\pgfplotmarksize + .5\pgflinewidth}%
	\pgfusepathqfill%
}

% filled square without border
\pgfdeclareplotmark{solidsquare}{%
	\pgfmathaddtolength{\pgfplotmarksize}{.5\pgflinewidth}%
  \pgfpathrectangle{\pgfqpoint{-\pgfplotmarksize}{-\pgfplotmarksize}}{\pgfqpoint{2\pgfplotmarksize}{2\pgfplotmarksize}}%
  \pgfusepathqfill%
}

% filled triangle without border
\pgfdeclareplotmark{solidtriangle}{%
	\pgfmathaddtolength{\pgfplotmarksize}{\pgflinewidth}%
  \pgfpathmoveto{\pgfqpoint{0pt}{\pgfplotmarksize}}%
  \pgfpathlineto{\pgfqpointpolar{-30}{\pgfplotmarksize}}%
  \pgfpathlineto{\pgfqpointpolar{-150}{\pgfplotmarksize}}%
  \pgfpathclose%
  \pgfusepathqfill%
}

% filled diamond without border
\pgfdeclareplotmark{soliddiamond}{%
	\pgfmathaddtolength{\pgfplotmarksize}{\pgflinewidth}%
  \pgfpathmoveto{\pgfqpoint{0pt}{\pgfplotmarksize}}%
  \pgfpathlineto{\pgfqpoint{.75\pgfplotmarksize}{0pt}}%
  \pgfpathlineto{\pgfqpoint{0pt}{-\pgfplotmarksize}}%
  \pgfpathlineto{\pgfqpoint{-.75\pgfplotmarksize}{0pt}}%
  \pgfpathclose%
  \pgfusepathqfill%
}

% plot cycle list for the filled markers without borders
\pgfplotscreateplotcyclelist{solid mark list}{mark=solido,mark=solidsquare,mark=solidtriangle,mark=soliddiamond}

% filled hexagon without border for use in hexbin plots
\pgfdeclareplotmark{hexagon}
{%
	\pgftransformyscale{\pggf@hex@yscale}
	\pgfmathsetlength{\pggf@tmplen}{0.5\pgfplotmarksize / cos(30)}
	\pgfpathmoveto{\pgfqpoint{0pt}{\pggf@tmplen}}
	\pgfpathlineto{\pgfqpointpolar{150}{\pggf@tmplen}}
	\pgfpathlineto{\pgfqpointpolar{210}{\pggf@tmplen}}
	\pgfpathlineto{\pgfqpointpolar{270}{\pggf@tmplen}}
	\pgfpathlineto{\pgfqpointpolar{330}{\pggf@tmplen}}
	\pgfpathlineto{\pgfqpointpolar{30}{\pggf@tmplen}}
	\pgfpathclose
	\pgfusepathqfill
}
